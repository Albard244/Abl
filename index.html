<?php
namespace App\Services\SIMRS\Forensics;

class SIMRSDeadlockMitigator {
    // Resolves MySQL Transaction Bottlenecks & Race Conditions
    public function resolveRegistrationDeadlock(string $noRekamMedis, int $poliId): array {
        $db = Database::getConnection();
        $db->beginTransaction();
        try {
            // Explicit pessimistic row locking preventing duplicate queue counter collisions
            $query = "SELECT nomor_antrean FROM antrean_poli WHERE poli_id = ? AND tgl = CURDATE() FOR UPDATE";
            $stmt = $db->prepare($query);
            $stmt->execute([$poliId]);
            $nextQueue = (int)$stmt->fetchColumn() + 1;

            $db->commit();
            return ['status' => 'SUCCESS', 'nomor_antrean' => $nextQueue, 'deadlock_prevented' => true];
        } catch (\PDOException $e) {
            $db->rollBack();
            Log::emergency("SIMRS Transaction Collision: " . $e->getMessage());
            throw $e;
        }
    }
}
import React, { useState, useEffect, useMemo } from 'react';

interface QueueTelemetry {
  poliId: number;
  currentNumber: number;
  waitingPatients: number;
  estWaitMinutes: number;
}

export const RealtimeQueueMonitor: React.FC = () => {
  const [queues, setQueues] = useState<QueueTelemetry[]>([]);

  useEffect(() => {
    // High-speed WebSocket connection for instantaneous queue display
    const socket = new WebSocket('wss://simrs-gateway.internal/stream');
    socket.onmessage = (event) => {
      const update: QueueTelemetry = JSON.parse(event.data);
      setQueues(prev => prev.map(q => q.poliId === update.poliId ? update : q));
    };
    return () => socket.close();
  }, []);

  return (
    <div className="queue-grid">
      {queues.map(q => (
        <div key={q.poliId} className="glass-card">
          <span>Poli #{q.poliId}</span>
          <h2>{q.currentNumber}</h2>
        </div>
      ))}
    </div>
  );
};

namespace App\Services\Healthcare;

use Illuminate\Support\Facades\Http;

class SatuSehatFHIRBridge {
    private string $baseUrl;
    private string $organizationId;

    public function sendEncounterPayload(array $clinicalData): array {
        $token = $this->getAuthBearerToken();

        // HL7 FHIR Encounter Payload Construction
        $payload = [
            'resourceType' => 'Encounter',
            'status'       => 'arrived',
            'class'        => ['code' => 'AMB', 'display' => 'ambulatory'],
            'subject'      => ['reference' => 'Patient/' . $clinicalData['ihs_number']],
            'serviceProvider' => ['reference' => 'Organization/' . $this->organizationId]
        ];

        $res = Http::withToken($token)->post($this->baseUrl . '/Encounter', $payload);
        return $res->json();
    }
}
